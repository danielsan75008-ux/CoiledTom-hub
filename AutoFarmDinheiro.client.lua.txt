    Coloque este LocalScript em:
        StarterPlayer > StarterPlayerScripts

        1. Crie ReplicatedStorage > Packages > WindUI (ModuleScript).
        2. Importe/sincronize o código-fonte oficial da WindUI nesse ModuleScript.
        3. Não use loadstring/game:HttpGet em produção.

    Estrutura esperada para o dinheiro (qualquer uma destas opções):
        Player.leaderstats.Dinheiro / Money / Cash
        Player.Data.Dinheiro / Money / Cash
        Um IntValue/NumberValue com um desses nomes dentro do Player
        Um atributo numérico Dinheiro / Money / Cash no Player ou em um descendente

    Este código usa somente APIs oficiais do Roblox. Para interagir com um prompt,
    ele usa ProximityPrompt:InputHoldBegin() e :InputHoldEnd(); não usa

        A compra, coleta e recompensa devem continuar sendo validadas pelo servidor
        nos scripts conectados a ProximityPrompt.Triggered.

    SCYTHE_PRICE = 2500,

    MONEY_NAMES = { "Dinheiro", "Money", "Cash" },

    SCYTHE_TOOL = "Foice",
    TRASH_TOOL = "Lixo",

    BUY_SCYTHE_ACTION = "Foice [Cortar Grama]",
    BUY_SCYTHE_OBJECT = "Comprar [2500$]",

    GRASS_ACTION = "Grama",
    GRASS_OBJECT = "Cortar [Requer Foice]",

    PICK_TRASH_ACTION = "Lixo",
    PICK_TRASH_OBJECT = "Recolher",

    TRASH_BIN_ACTION = "Lixeira",
    TRASH_BIN_OBJECT = "Jogar Lixo",

    SEARCH_DELAY = 0.20,
    TELEPORT_SETTLE_DELAY = 0.18,
    AFTER_INTERACTION_DELAY = 0.25,
    POLL_DELAY = 0.05,
    TOOL_TIMEOUT = 8,
    PROMPT_REUSE_DELAY = 1.00,
}

-- Referências preenchidas depois da criação dos Toggles.
local GrassToggle
local TrashToggle

local function notify(title, content, icon)
    pcall(function()
        WindUI:Notify({
            Title = title,
            Content = content,
            Icon = icon or "info",
            Duration = 5,
            CanClose = true,
        })
    end)
end

-- =========================================================
-- Controle cancelável: uma única thread por Toggle
-- =========================================================

local function newFarmState(name)
    return {
        name = name,
        enabled = false,
        revision = 0,
        wakeEvent = Instance.new("BindableEvent"),
    }
end

local GrassState = newFarmState("Auto Farm Grama")
local TrashState = newFarmState("Auto Farm Lixo")

local function setFarmEnabled(state, enabled)
    enabled = enabled == true

    if state.enabled == enabled then
        return
    end

    state.enabled = enabled
    state.revision = state.revision + 1
    state.wakeEvent:Fire()
end

-- Espera em pequenos intervalos e cancela assim que o Toggle muda de estado.
local function cancellableWait(duration, isActive)
    local deadline = os.clock() + math.max(duration or 0, 0)

    repeat
        if not isActive() then
            return false
        end

        local remaining = deadline - os.clock()
        if remaining <= 0 then
            break
        end

        task.wait(math.min(CONFIG.POLL_DELAY, remaining))
    until false

    return isActive()
end

-- =========================================================
-- Character, ferramentas e dinheiro
-- =========================================================

local function getCharacterParts()
    local character = LocalPlayer.Character
    if not character or not character.Parent then
        return nil, nil, nil
    end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local rootPart = character:FindFirstChild("HumanoidRootPart")

    if not humanoid or humanoid.Health <= 0 or not rootPart then
        return nil, nil, nil
    end

    return character, humanoid, rootPart
end

-- Procura uma Tool no Character e no Backpack.
local function findTool(toolName)
    local character = LocalPlayer.Character
    if character then
        local equippedTool = character:FindFirstChild(toolName)
        if equippedTool and equippedTool:IsA("Tool") then
            return equippedTool
        end
    end

    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    if backpack then
        local storedTool = backpack:FindFirstChild(toolName)
        if storedTool and storedTool:IsA("Tool") then
            return storedTool
        end
    end

    return nil
end

-- Equipa uma Tool do Backpack. Se já estiver no Character, não faz nada.
local function equipTool(tool, isActive)
    if not tool or not tool.Parent or not isActive() then
        return false
    end

    local character, humanoid = getCharacterParts()
    if not character or not humanoid then
        return false
    end

    if tool.Parent == character then
        return true
    end

    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    if not backpack or tool.Parent ~= backpack then
        return false
    end

    local success = pcall(function()
        humanoid:EquipTool(tool)
    end)

    if not success then
        return false
    end

    local deadline = os.clock() + 1.5
    while os.clock() < deadline do
        if not isActive() then
            return false
        end

        if tool.Parent == character then
            return true
        end

        task.wait(CONFIG.POLL_DELAY)
    end

    return tool.Parent == character
end

-- Espera uma Tool aparecer ou desaparecer sem usar um WaitForChild infinito.
local function waitForToolState(toolName, shouldExist, timeout, isActive)
    local deadline = os.clock() + timeout

    repeat
        if not isActive() then
            return false, nil
        end

        local tool = findTool(toolName)
        if (tool ~= nil) == shouldExist then
            return true, tool
        end

        task.wait(CONFIG.POLL_DELAY)
    until os.clock() >= deadline

    return false, findTool(toolName)
end

local function isMoneyName(name)
    for _, acceptedName in ipairs(CONFIG.MONEY_NAMES) do
        if name == acceptedName then
            return true
        end
    end

    return false
end

-- Localiza o dinheiro em ValueObjects ou atributos numéricos do Player.
local function getMoneyAmount()
    local leaderstats = LocalPlayer:FindFirstChild("leaderstats")

    if leaderstats then
        for _, moneyName in ipairs(CONFIG.MONEY_NAMES) do
            local valueObject = leaderstats:FindFirstChild(moneyName)
            if valueObject and (valueObject:IsA("IntValue") or valueObject:IsA("NumberValue")) then
                return tonumber(valueObject.Value), valueObject
            end
        end
    end

    for _, descendant in ipairs(LocalPlayer:GetDescendants()) do
        if isMoneyName(descendant.Name)
            and (descendant:IsA("IntValue") or descendant:IsA("NumberValue")) then
            return tonumber(descendant.Value), descendant
        end
    end

    local containers = { LocalPlayer }
    for _, descendant in ipairs(LocalPlayer:GetDescendants()) do
        table.insert(containers, descendant)
    end

    for _, container in ipairs(containers) do
        for _, moneyName in ipairs(CONFIG.MONEY_NAMES) do
            local attributeValue = container:GetAttribute(moneyName)
            if typeof(attributeValue) == "number" then
                return attributeValue, container
            end
        end
    end

    return nil, nil
end

-- =========================================================
-- Registro e busca de ProximityPrompts por todo o Workspace
-- =========================================================

-- Mantém apenas os prompts em um índice, evitando GetDescendants() a cada ciclo.
local PromptRegistry = {}
local PromptCooldowns = setmetatable({}, { __mode = "k" })

local function registerPrompt(instance)
    if instance:IsA("ProximityPrompt") then
        PromptRegistry[instance] = true
    end
end

for _, descendant in ipairs(Workspace:GetDescendants()) do
    registerPrompt(descendant)
end

Workspace.DescendantAdded:Connect(registerPrompt)
Workspace.DescendantRemoving:Connect(function(instance)
    if instance:IsA("ProximityPrompt") then
        PromptRegistry[instance] = nil
        PromptCooldowns[instance] = nil
    end
end)

-- Compara exatamente ActionText e ObjectText; nunca depende do Name do objeto.
local function promptMatches(prompt, actionText, objectText)
    return prompt
        and prompt:IsA("ProximityPrompt")
        and prompt.Parent ~= nil
        and prompt:IsDescendantOf(Workspace)
        and prompt.Enabled
        and prompt.ActionText == actionText
        and prompt.ObjectText == objectText
end

-- Obtém a posição de um Prompt mesmo quando ele está em Attachment/Part/Model.
local function getPromptCFrame(prompt)
    if not prompt or not prompt.Parent then
        return nil
    end

    local parent = prompt.Parent

    if parent:IsA("Attachment") then
        return parent.WorldCFrame
    end

    if parent:IsA("BasePart") then
        return parent.CFrame
    end

    local attachment = prompt:FindFirstAncestorWhichIsA("Attachment")
    if attachment then
        return attachment.WorldCFrame
    end

    local basePart = prompt:FindFirstAncestorWhichIsA("BasePart")
    if basePart then
        return basePart.CFrame
    end

    local model = prompt:FindFirstAncestorWhichIsA("Model")
    if model and model:IsDescendantOf(Workspace) then
        local success, pivot = pcall(function()
            return model:GetPivot()
        end)

        if success then
            return pivot
        end
    end

    return nil
end

-- Retorna o Prompt correspondente mais próximo. Prompts usados recentemente
-- recebem um cooldown curto para não serem acionados várias vezes em sequência.
local function findPrompt(actionText, objectText)
    local _, _, rootPart = getCharacterParts()
    local rootPosition = rootPart and rootPart.Position or nil
    local now = os.clock()

    local bestPrompt = nil
    local bestDistance = math.huge

    for prompt in pairs(PromptRegistry) do
        if promptMatches(prompt, actionText, objectText)
            and (PromptCooldowns[prompt] or 0) <= now then

            local promptCFrame = getPromptCFrame(prompt)
            if promptCFrame then
                local distance = 0
                if rootPosition then
                    distance = (rootPosition - promptCFrame.Position).Magnitude
                end

                if distance < bestDistance then
                    bestDistance = distance
                    bestPrompt = prompt
                end
            end
        end
    end

    return bestPrompt
end

local function putPromptOnCooldown(prompt)
    if prompt then
        PromptCooldowns[prompt] = os.clock() + CONFIG.PROMPT_REUSE_DELAY
    end
end

-- =========================================================
-- Teleporte e acionamento oficial do ProximityPrompt
-- =========================================================

-- Evita que os dois farms tentem teleportar o Character ao mesmo tempo.
local InteractionOwner = nil

local function acquireInteraction(owner, isActive)
    while InteractionOwner ~= nil and InteractionOwner ~= owner do
        if not cancellableWait(CONFIG.POLL_DELAY, isActive) then
            return false
        end
    end

    if not isActive() then
        return false
    end

    InteractionOwner = owner
    return true
end

local function releaseInteraction(owner)
    if InteractionOwner == owner then
        InteractionOwner = nil
    end
end

-- Teleporta o Character para dentro da distância permitida pelo Prompt.
local function teleportToPrompt(prompt, isActive)
    if not isActive() or not prompt or not prompt.Parent then
        return false
    end

    local promptCFrame = getPromptCFrame(prompt)
    local character, _, rootPart = getCharacterParts()

    if not promptCFrame or not character or not rootPart then
        return false
    end

    local maxDistance = math.max(prompt.MaxActivationDistance, 1)
    local verticalOffset = math.clamp(maxDistance * 0.30, 0.35, 3)
    local targetPosition = promptCFrame.Position + Vector3.new(0, verticalOffset, 0)
    local targetCFrame = CFrame.lookAt(targetPosition, promptCFrame.Position)

    local success = pcall(function()
        character:PivotTo(targetCFrame)
        rootPart.AssemblyLinearVelocity = Vector3.zero
        rootPart.AssemblyAngularVelocity = Vector3.zero
    end)

    if not success then
        return false
    end

    return cancellableWait(CONFIG.TELEPORT_SETTLE_DELAY, isActive)
end

-- Aciona o Prompt por APIs oficiais. Se o Toggle for desligado durante o hold,
-- InputHoldEnd() é chamado e a interação é cancelada imediatamente.
local function activatePrompt(prompt, actionText, objectText, isActive)
    if not isActive() or not promptMatches(prompt, actionText, objectText) then
        return false
    end

    local began = pcall(function()
        prompt:InputHoldBegin()
    end)

    if not began then
        return false
    end

    local completed = cancellableWait(math.max(prompt.HoldDuration, 0) + 0.10, isActive)

    -- O Prompt pode ter sido destruído pela própria interação; por isso, pcall.
    pcall(function()
        prompt:InputHoldEnd()
    end)

    return completed
end

-- Combina lock, teleporte e ativação, sempre liberando o lock no final.
local function usePrompt(prompt, actionText, objectText, owner, isActive)
    if not acquireInteraction(owner, isActive) then
        return false
    end

    local success, result = xpcall(function()
        if not promptMatches(prompt, actionText, objectText) then
            return false
        end

        if not teleportToPrompt(prompt, isActive) then
            return false
        end

        -- Confere novamente porque o Prompt pode desaparecer durante o teleporte.
        if not promptMatches(prompt, actionText, objectText) then
            return false
        end

        return activatePrompt(prompt, actionText, objectText, isActive)
    end, debug.traceback)

    releaseInteraction(owner)

    if not success then
        warn("Falha ao usar ProximityPrompt:\n" .. tostring(result))
        return false
    end

    return result == true
end

-- =========================================================
-- Auto Farm Grama
-- =========================================================

local function stopGrassToggle()
    setFarmEnabled(GrassState, false)

    -- Mantém a parte visual da WindUI sincronizada com o estado interno.
    task.defer(function()
        if GrassToggle then
            pcall(function()
                GrassToggle:Set(false)
            end)
        end
    end)
end

local function validateGrassMoney()
    local money = getMoneyAmount()

    if money == nil then
        notify(
            "Auto Farm Grama",
            "Não foi possível localizar o dinheiro do jogador.",
            "triangle-alert"
        )
        stopGrassToggle()
        return false
    end

    if money < CONFIG.SCYTHE_PRICE then
        notify(
            "Auto Farm Grama",
            "Por favor, use o Auto Farm Lixo para pegar o dinheiro necessário.",
            "triangle-alert"
        )
        stopGrassToggle()
        return false
    end

    return true
end

local function grassFarmSession(isActive)
    -- Conforme solicitado, a validação de 2500$ ocorre antes de qualquer compra.
    if not validateGrassMoney() or not isActive() then
        return
    end

    local scythe = findTool(CONFIG.SCYTHE_TOOL)

    -- Compra a Foice apenas se ela não estiver no Backpack/Character.
    while isActive() and not scythe do
        local purchasePrompt = findPrompt(
            CONFIG.BUY_SCYTHE_ACTION,
            CONFIG.BUY_SCYTHE_OBJECT
        )

        if purchasePrompt then
            local used = usePrompt(
                purchasePrompt,
                CONFIG.BUY_SCYTHE_ACTION,
                CONFIG.BUY_SCYTHE_OBJECT,
                GrassState,
                isActive
            )

            if used then
                putPromptOnCooldown(purchasePrompt)
                local appeared, foundTool = waitForToolState(
                    CONFIG.SCYTHE_TOOL,
                    true,
                    CONFIG.TOOL_TIMEOUT,
                    isActive
                )

                if appeared then
                    scythe = foundTool
                else
                    -- Se o dinheiro foi consumido sem entregar a Tool, evita
                    -- continuar tentando uma compra impossível.
                    local currentMoney = getMoneyAmount()
                    if currentMoney ~= nil and currentMoney < CONFIG.SCYTHE_PRICE then
                        notify(
                            "Auto Farm Grama",
                            "Por favor, use o Auto Farm Lixo para pegar o dinheiro necessário.",
                            "triangle-alert"
                        )
                        stopGrassToggle()
                        return
                    end
                end
            end
        else
            cancellableWait(CONFIG.SEARCH_DELAY, isActive)
        end
    end

    if not isActive() or not scythe then
        return
    end

    -- Se estiver no Backpack, equipa diretamente.
    while isActive() and not equipTool(scythe, isActive) do
        scythe = findTool(CONFIG.SCYTHE_TOOL)
        if not scythe then
            return
        end
        cancellableWait(CONFIG.SEARCH_DELAY, isActive)
    end

    -- Procura e corta gramas continuamente enquanto o Toggle permanecer ligado.
    while isActive() do
        scythe = findTool(CONFIG.SCYTHE_TOOL)

        if not scythe then
            -- Pode acontecer durante respawn ou caso a Tool seja removida.
            local appeared, foundTool = waitForToolState(
                CONFIG.SCYTHE_TOOL,
                true,
                3,
                isActive
            )

            if not appeared then
                return
            end

            scythe = foundTool
        end

        if scythe and scythe.Parent ~= LocalPlayer.Character then
            equipTool(scythe, isActive)
        end

        local grassPrompt = findPrompt(CONFIG.GRASS_ACTION, CONFIG.GRASS_OBJECT)

        if grassPrompt then
            local used = usePrompt(
                grassPrompt,
                CONFIG.GRASS_ACTION,
                CONFIG.GRASS_OBJECT,
                GrassState,
                isActive
            )

            if used then
                putPromptOnCooldown(grassPrompt)
            end

            cancellableWait(CONFIG.AFTER_INTERACTION_DELAY, isActive)
        else
            cancellableWait(CONFIG.SEARCH_DELAY, isActive)
        end
    end
end

-- =========================================================
-- Auto Farm Lixo
-- =========================================================

local function trashFarmSession(isActive)
    while isActive() do
        local trashTool = findTool(CONFIG.TRASH_TOOL)

        if trashTool then
            -- Com Lixo no inventário, procura uma lixeira.
            local binPrompt = findPrompt(
                CONFIG.TRASH_BIN_ACTION,
                CONFIG.TRASH_BIN_OBJECT
            )

            if binPrompt then
                local used = usePrompt(
                    binPrompt,
                    CONFIG.TRASH_BIN_ACTION,
                    CONFIG.TRASH_BIN_OBJECT,
                    TrashState,
                    isActive
                )

                if used then
                    putPromptOnCooldown(binPrompt)

                    -- Aguarda o servidor remover a Tool Lixo.
                    waitForToolState(
                        CONFIG.TRASH_TOOL,
                        false,
                        CONFIG.TOOL_TIMEOUT,
                        isActive
                    )
                end

                cancellableWait(CONFIG.AFTER_INTERACTION_DELAY, isActive)
            else
                cancellableWait(CONFIG.SEARCH_DELAY, isActive)
            end
        else
            -- Sem Lixo, procura um ponto de coleta.
            local pickupPrompt = findPrompt(
                CONFIG.PICK_TRASH_ACTION,
                CONFIG.PICK_TRASH_OBJECT
            )

            if pickupPrompt then
                local used = usePrompt(
                    pickupPrompt,
                    CONFIG.PICK_TRASH_ACTION,
                    CONFIG.PICK_TRASH_OBJECT,
                    TrashState,
                    isActive
                )

                if used then
                    putPromptOnCooldown(pickupPrompt)

                    -- Aguarda o servidor colocar a Tool no Backpack/Character.
                    waitForToolState(
                        CONFIG.TRASH_TOOL,
                        true,
                        CONFIG.TOOL_TIMEOUT,
                        isActive
                    )
                end

                cancellableWait(CONFIG.AFTER_INTERACTION_DELAY, isActive)
            else
                cancellableWait(CONFIG.SEARCH_DELAY, isActive)
            end
        end
    end
end

-- =========================================================
-- Worker único por farm
-- =========================================================

local function startSingleWorker(state, sessionFunction)
    task.spawn(function()
        while true do
            -- Quando desligado, a thread fica suspensa e não faz polling.
            if not state.enabled then
                state.wakeEvent.Event:Wait()
            else
                local sessionRevision = state.revision

                local function isActive()
                    return state.enabled and state.revision == sessionRevision
                end

                local success, errorMessage = xpcall(function()
                    sessionFunction(isActive)
                end, debug.traceback)

                if not success then
                    warn(state.name .. " encontrou um erro:\n" .. tostring(errorMessage))

                    if isActive() then
                        notify(
                            state.name,
                            "A automação encontrou um erro e tentará novamente.",
                            "triangle-alert"
                        )
                    end
                end

                -- Evita reinício apertado se uma sessão terminar por falta de
                -- Character/Tool. A espera continua sendo cancelável.
                if isActive() then
                    cancellableWait(CONFIG.SEARCH_DELAY, isActive)
                end
            end
        end
    end)
end

TrashToggle = MoneyTab:Toggle({
    Title = "Auto Farm Lixo",

    Type = "Toggle",
    Value = false,

startSingleWorker(GrassState, grassFarmSession)
startSingleWorker(TrashState, trashFarmSession)
